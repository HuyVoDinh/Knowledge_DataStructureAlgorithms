void func(int n)    -> T(n) time
{
    if (n>0)        -> 1 time
    {
        cout << n << endl; -> 1 time
        func(n-1);  -> T(n-1)
    }
}
-> T(n) = T(n-1) + 2    (n > 0)
         = 1             (n = 0)

T(n) = T(n-1) + 1   but T(n-1) = T(n-2) + 1
-> T(n) = T(n-2) + 2
.....
-> T(n) = T(n-k) + k 
Assume n - k = 0: n = k 
T(n) = T(n-n) + n 
T(n) = T(0) + n
T(n) = 1 + n
=> O(n)
